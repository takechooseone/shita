local function getrole(p)
    if not p or not p.Character then return "?", Color3.fromRGB(180, 180, 180) end
    local tool = p.Character:FindFirstChildOfClass("Tool")
    if tool then
        local n = tool.Name:lower()
        if n:find("knife") or n:find("dagger") then
            return "MURDERER", Color3.fromRGB(255, 60, 60)
        end
        if n:find("gun") or n:find("pistol") or n:find("revolver") then
            return "SHERIFF", Color3.fromRGB(60, 140, 255)
        end
    end
    local ls = p:FindFirstChild("leaderstats")
    if ls then
        for _, v in ipairs(ls:GetChildren()) do
            local val = tostring(v.Value):lower()
            if val:find("murder") then return "MURDERER", Color3.fromRGB(255, 60, 60) end
            if val:find("sheriff") then return "SHERIFF", Color3.fromRGB(60, 140, 255) end
            if val:find("innocent") then return "INNOCENT", Color3.fromRGB(80, 220, 120) end
        end
    end
    return "INNOCENT", Color3.fromRGB(80, 220, 120)
end

local function getclosest(rad)
    local closest, dist = nil, rad or math.huge
    local mypos = plr.Character and plr.Character.HumanoidRootPart and plr.Character.HumanoidRootPart.Position
    if not mypos then return nil end
    for _, p in ipairs(game.Players:GetPlayers()) do
        if p ~= plr and p.Character then
            local hrp = p.Character:FindFirstChild("HumanoidRootPart")
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            if hrp and hum and hum.Health > 0 then
                local d = (hrp.Position - mypos).Magnitude
                if d < dist then dist = d; closest = p end
            end
        end
    end
    return closest
end

local function silentaim()
    if not state.silent then return end
    local target = getclosest(500)
    if target and target.Character and target.Character:FindFirstChild("Head") then
        local head = target.Character.Head
        local hrp = target.Character:FindFirstChild("HumanoidRootPart")
        if hrp then
            local vel = hrp.AssemblyLinearVelocity or Vector3.new()
            local dist = (head.Position - cam.CFrame.Position).Magnitude
            local t = dist / 200
            local pos = head.Position + vel * t * state.prediction
            cam.CFrame = CFrame.lookAt(cam.CFrame.Position, pos)
        end
    end
end

local function killaura()
    if not state.killaura then return end
    if tick() - aura_last < state.aura_delay then return end
    aura_last = tick()
    local target = getclosest(state.aura_radius)
    if not target then return end
    local tool = plr.Character and plr.Character:FindFirstChildOfClass("Tool")
    if tool then pcall(function() tool:Activate() end) end
end

local function getweapons()
    local folder = workspace:FindFirstChild("Weapons") or workspace:FindFirstChild("Ignored")
    if not folder then return {} end
    local list = {}
    for _, g in ipairs(folder:GetChildren()) do
        if g:FindFirstChild("Handle") then table.insert(list, g) end
    end
    return list
end

local function autofarm()
    if not state.autofarm then return end
    local hrp = plr.Character and plr.Character.HumanoidRootPart
    if not hrp then return end
    local weapons = getweapons()
    if #weapons == 0 then return end
    local closest, dist = nil, math.huge
    for _, g in ipairs(weapons) do
        local d = (g.Handle.Position - hrp.Position).Magnitude
        if d < dist then dist = d; closest = g end
    end
    if closest then
        local dir = (closest.Handle.Position - hrp.Position)
        hrp.CFrame = hrp.CFrame + dir.Unit * state.autofarm_speed
    end
end

local function cleanup_esp(p)
    local data = esp_cache[p]
    if data then
        pcall(function()
            if data.hl then data.hl:Destroy() end
            if data.bg then data.bg:Destroy() end
        end)
        esp_cache[p] = nil
    end
end

local function cleanup_all_esp()
    for p, _ in pairs(esp_cache) do
        cleanup_esp(p)
    end
end

local function esp()
    if not state.esp then
        cleanup_all_esp()
        return
    end
    for _, p in ipairs(game.Players:GetPlayers()) do
        if p ~= plr and p.Character then
            local data = esp_cache[p]
            if not data or not data.hl or not data.hl.Parent or not data.bg or not data.bg.Parent then
                cleanup_esp(p)

                hl.FillColor = colors.main
                hl.OutlineColor = Color3.fromRGB(255, 255, 255)
                hl.FillTransparency = 0.55
                hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                hl.Parent = p.Character

                local head = p.Character:FindFirstChild("Head")
                local bg = nil
                if head then
                    bg = Instance.new("BillboardGui")
                    bg.Size = UDim2.new(0, 240, 0, 80)
                    bg.StudsOffset = Vector3.new(0, 3, 0)
                    bg.AlwaysOnTop = true
                    bg.Parent = head

                    local role = Instance.new("TextLabel")
                    role.Name = "role"
                    role.Size = UDim2.new(1, 0, 0, 22)
                    role.BackgroundTransparency = 1
                    role.TextColor3 = Color3.fromRGB(255, 255, 255)
                    role.TextStrokeTransparency = 0
                    role.Font = Enum.Font.GothamBold
                    role.TextSize = 15
                    role.Text = "?"
                    role.Parent = bg

                    local nm = Instance.new("TextLabel")
                    nm.Name = "nm"
                    nm.Size = UDim2.new(1, 0, 0, 20)
                    nm.Position = UDim2.new(0, 0, 0, 22)
                    nm.BackgroundTransparency = 1
                    nm.TextColor3 = Color3.fromRGB(255, 255, 255)
                    nm.TextStrokeTransparency = 0
                    nm.Font = Enum.Font.Gotham
                    nm.TextSize = 13
                    nm.Text = p.Name
                    nm.Parent = bg

                    local hpl = Instance.new("TextLabel")
                    hpl.Name = "hp"
                    hpl.Size = UDim2.new(1, 0, 0, 18)
                    hpl.Position = UDim2.new(0, 0, 0, 42)
                    hpl.BackgroundTransparency = 1
                    hpl.TextColor3 = Color3.fromRGB(80, 220, 120)
                    hpl.TextStrokeTransparency = 0
                    hpl.Font = Enum.Font.Gotham
                    hpl.TextSize = 12
                    hpl.Parent = bg

                    local dl = Instance.new("TextLabel")
                    dl.Name = "dl"
                    dl.Size = UDim2.new(1, 0, 0, 18)
                    dl.Position = UDim2.new(0, 0, 0, 60)
                    dl.BackgroundTransparency = 1
                    dl.TextColor3 = Color3.fromRGB(200, 200, 200)
                    dl.TextStrokeTransparency = 0
                    dl.Font = Enum.Font.Gotham
                    dl.TextSize = 12
                    dl.Parent = bg
                end
                esp_cache[p] = {hl = hl, bg = bg}
                data = esp_cache[p]
            end
            if data and data.bg and data.hl and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
                local hum = p.Character:FindFirstChildOfClass("Humanoid")
                local hrp = p.Character:FindFirstChild("HumanoidRootPart")
                if hum and hrp then
                    local dist = (hrp.Position - plr.Character.HumanoidRootPart.Position).Magnitude
                    local roletext, rolecolor = getrole(p)
                    local role = data.bg:FindFirstChild("role")
                    local nm = data.bg:FindFirstChild("nm")
                    local hpl = data.bg:FindFirstChild("hp")
                    local dl = data.bg:FindFirstChild("dl")
                    if role then
                        role.Text = state.esp_role and roletext or ""
                        role.TextColor3 = rolecolor
                    end
                    if nm then nm.Text = state.esp_name and p.Name or "" end
                    if hpl then hpl.Text = state.esp_health and string.format("hp: %d", math.floor(hum.Health)) or "" end
                    if dl then dl.Text = state.esp_distance and string.format("%d studs", math.floor(dist)) or "" end
                    if data.hl then data.hl.FillColor = rolecolor end
                end
            end
        end
    end
    for p, data in pairs(esp_cache) do
        if not p.Character or not p.Character.Parent then
            cleanup_esp(p)
        end
    end
end

local function cleanup_gun(tool)
    local data = gun_cache[tool]
    if data then
        pcall(function()
            if data.hl then data.hl:Destroy() end
            if data.bg then data.bg:Destroy() end
            if data.parts then
                for part, orig in pairs(data.parts) do
                    if part and part.Parent then part.Transparency = orig end
                end
            end
        end)
        gun_cache[tool] = nil
    end
end

local function cleanup_all_gun()
    for tool, _ in pairs(gun_cache) do
        cleanup_gun(tool)
    end
end

local function gunesp()
    if not state.gunesp then
        cleanup_all_gun()
        return
    end
    local theme = gune_colors[state.gunesp_color] or gune_colors.purple
    for _, p in ipairs(game.Players:GetPlayers()) do
        if p ~= plr and p.Character then
            local tool = p.Character:FindFirstChildOfClass("Tool")
            if tool then
                local n = tool.Name:lower()
                local isknife = n:find("knife") or n:find("dagger")
                local isgun = n:find("gun") or n:find("pistol") or n:find("revolver")
                local handle = tool:FindFirstChild("Handle")

                if handle and (isknife or isgun) then
                    local data = gun_cache[tool]
                    if not data or not data.hl or not data.hl.Parent then
                        cleanup_gun(tool)
                        local hl = Instance.new("Highlight")
                        hl.FillTransparency = 1
                        hl.OutlineColor = theme.outline
                        hl.OutlineTransparency = 0
                        hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                        hl.Adornee = tool
                        hl.Parent = tool

                        local bg = Instance.new("BillboardGui")
                        bg.Size = UDim2.new(0, 180, 0, 20)
                        bg.StudsOffset = Vector3.new(0, 1.5, 0)
                        bg.AlwaysOnTop = true
                        bg.Parent = handle

                        local lbl = Instance.new("TextLabel")
                        lbl.Size = UDim2.new(1, 0, 1, 0)
                        lbl.BackgroundTransparency = 1
                        lbl.TextStrokeTransparency = 0
                        lbl.Font = Enum.Font.GothamBold
                        lbl.TextSize = 13
                        lbl.TextColor3 = theme.outline
                        lbl.Text = p.Name .. " | " .. tool.Name
                        lbl.Parent = bg

                        local parts = {}
                        for _, obj in ipairs(tool:GetDescendants()) do
                            if obj:IsA("BasePart") then parts[obj] = obj.Transparency end
                        end

                        gun_cache[tool] = {hl = hl, bg = bg, parts = parts}
                        data = gun_cache[tool]
                    else
                        if data.hl then data.hl.OutlineColor = theme.outline end
                        if data.bg and data.bg:FindFirstChildOfClass("TextLabel") then
                            data.bg:FindFirstChildOfClass("TextLabel").TextColor3 = theme.outline
                        end
                    end

                    if data and data.parts then
                        for part, orig in pairs(data.parts) do
                            if part and part.Parent then
                                part.Transparency = math.max(orig, GUN_TRANSPARENCY)
                                part.CanCollide = false
                            end
                        end
                    end
                end
            end
        end
    end
    for tool, data in pairs(gun_cache) do
        if not tool or not tool.Parent then cleanup_gun(tool) end
    end
end

local function fullbright()
    if state.fullbright then
        lighting.Ambient = Color3.fromRGB(200, 200, 200)
        lighting.Brightness = 2
        lighting.ClockTime = 12
        lighting.FogEnd = 100000
        lighting.GlobalShadows = false
    end
end

local function noclip()
    if not state.noclip then return end
    if plr.Character then
        for _, p in ipairs(plr.Character:GetDescendants()) do
            if p:IsA("BasePart") then p.CanCollide = false end
        end
    end
end

local function antifling()
    if not state.anti_fling then return end
    if plr.Character then
        local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
        if hrp and hrp.AssemblyLinearVelocity.Magnitude > 200 then
            hrp.AssemblyLinearVelocity = Vector3.new()
            hrp.AssemblyAngularVelocity = Vector3.new()
        end
    end
end

local function antivoid()
    if not state.anti_void then return end
    if plr.Character then
        local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
        if hrp and hrp.Position.Y < -50 then hrp.CFrame = CFrame.new(0, 50, 0) end
    end
end

local function applyspeed()
    if plr.Character and plr.Character:FindFirstChildOfClass("Humanoid") then
        plr.Character.Humanoid.WalkSpeed = state.walkspeed
        plr.Character.Humanoid.JumpPower = state.jumppower
        plr.Character.Humanoid.UseJumpPower = true
    end
end

local tabs = {}

local function selecttab(name)
    for tn, td in pairs(tabs) do
        if tn == name then
            td.button.BackgroundColor3 = colors.on
            td.button.TextColor3 = colors.main
            td.button.Font = Enum.Font.GothamBold
        else
            td.button.BackgroundColor3 = colors.bg3
            td.button.TextColor3 = colors.textdim
            td.button.Font = Enum.Font.Gotham
        end
    end
    for _, child in ipairs(content:GetChildren()) do
        if child:IsA("GuiObject") and child.Name:sub(1, 6) == "tab_c_" then
            child.Visible = child.Name == "tab_c_" .. name
        end
    end
end

local function createtab(name)
    local button = Instance.new("TextButton")
    button.Size = UDim2.new(0, 100, 1, 0)
    button.BackgroundColor3 = colors.bg3
    button.BorderSizePixel = 0
    button.Text = name
    button.TextColor3 = colors.textdim
    button.Font = Enum.Font.Gotham
    button.TextSize = 13
    button.Parent = tabbar

    local bc = Instance.new("UICorner")
    bc.CornerRadius = UDim.new(0, 8)
    bc.Parent = button

    local container = Instance.new("Frame")
    container.Name = "tab_c_" .. name
    container.Size = UDim2.new(1, 0, 0, 0)
    container.AutomaticSize = Enum.AutomaticSize.Y
    container.BackgroundTransparency = 1
    container.BorderSizePixel = 0
    container.Visible = false
    container.Parent = content

    local cl = Instance.new("UIListLayout")
    cl.Padding = UDim.new(0, 8)
    cl.SortOrder = Enum.SortOrder.LayoutOrder
    cl.Parent = container

    button.MouseButton1Click:Connect(function() selecttab(name) end)

    tabs[name] = {button = button, container = container, layout = cl}
    return container
end

local function makeslider(parent, name, key, min, max, def)
    local holder = Instance.new("Frame")
    holder.Size = UDim2.new(1, 0, 0, 56)
    holder.BackgroundColor3 = colors.bg3
    holder.BorderSizePixel = 0
    holder.Parent = parent

    local hc = Instance.new("UICorner")
    hc.CornerRadius = UDim.new(0, 8)
    hc.Parent = holder

    local lbl = Instance.new("TextLabel")
    lbl.Size = UDim2.new(1, -20, 0, 22)
    lbl.Position = UDim2.new(0, 12, 0, 4)
    lbl.BackgroundTransparency = 1
    lbl.Text = name .. "   " .. string.format("%.2f", def)
    lbl.TextColor3 = colors.text
    lbl.Font = Enum.Font.Gotham
    lbl.TextSize = 13
    lbl.TextXAlignment = Enum.TextXAlignment.Left
    lbl.Parent = holder

    local bar = Instance.new("Frame")
    bar.Size = UDim2.new(1, -24, 0, 8)
    bar.Position = UDim2.new(0, 12, 0, 34)
    bar.BackgroundColor3 = Color3.fromRGB(50, 50, 60)
    bar.BorderSizePixel = 0
    bar.Parent = holder

    local bc = Instance.new("UICorner")
    bc.CornerRadius = UDim.new(1, 0)
    bc.Parent = bar

    local fill = Instance.new("Frame")
    fill.Size = UDim2.new((def - min) / (max - min), 0, 1, 0)
    fill.BackgroundColor3 = colors.main
    fill.BorderSizePixel = 0
    fill.Parent = bar

    local fc = Instance.new("UICorner")
    fc.CornerRadius = UDim.new(1, 0)
    fc.Parent = fill

    local dragging = false

    local function update(input)
        local rel = math.clamp((input.Position.X - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
        local val = min + (max - min) * rel
        fill.Size = UDim2.new(rel, 0, 1, 0)
        lbl.Text = name .. "   " .. string.format("%.2f", val)
        state[key] = val
    end

    bar.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            update(i)
        end
    end)
    bar.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    uis.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            update(i)
        end
    end)
    uis.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end

local function makecycle(parent, name, key, options)
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(1, 0, 0, 34)
    btn.BackgroundColor3 = colors.bg3
    btn.BorderSizePixel = 0
    btn.Text = name .. "   " .. tostring(state[key])
    btn.TextColor3 = colors.text
    btn.Font = Enum.Font.Gotham
    btn.TextSize = 14
    btn.TextXAlignment = Enum.TextXAlignment.Left
    btn.Parent = parent

    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, 8)
    c.Parent = btn

    local pad = Instance.new("UIPadding")
    pad.PaddingLeft = UDim.new(0, 12)
    pad.Parent = btn

    local idx = 1
    for i, o in ipairs(options) do
        if o == state[key] then idx = i end
    end

    btn.MouseButton1Click:Connect(function()
        idx = idx + 1
        if idx > #options then idx = 1 end
        state[key] = options[idx]
        btn.Text = name .. "   " .. tostring(options[idx])
    end)
end
