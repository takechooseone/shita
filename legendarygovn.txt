local function loadmain()
    repeat task.wait() until game:IsLoaded()
    repeat task.wait() until plr.Character
    repeat task.wait() until plr.Character:FindFirstChild("HumanoidRootPart")

    local function gethitsound()
        if hitsound then return hitsound end
        local s = instance.new("Sound")
        s.SoundId = "rbxassetid://131961136"
        s.Volume = 0.5
        s.Parent = sg
        hitsound = s
        return s
    end

    local function playhitsound()
        if not state.silent_hitsound then return end
        local s = gethitsound()
        if s then
            pcall(function() s:Play() end)
        end
    end

    local mt = getrawmetatable(game)
    local oldnamecall = mt.__namecall
    setreadonly(mt, false)
    mt.__namecall = newcclosure(function(self, ...)
        local method = getnamecallmethod()
        local args = {...}
        if method == "Kick" then
            notify("anti-kick", "kick blocked")
            return
        end
        if method == "FireServer" then
            if state.silent and (self.Name:lower():find("knife") or self.Name:lower():find("swing") or self.Name:lower():find("attack")) then
                local target = state.silent_target
                if target and target.Character and target.Character:FindFirstChild("HumanoidRootPart") then
                    args[1] = target.Character.HumanoidRootPart
                    playhitsound()
                end
            end
            return oldnamecall(self, table.unpack(args))
        end
        return oldnamecall(self, ...)
    end)
    setreadonly(mt, true)

    local function getclosest(rad)
        rad = rad or math.huge
        local closest, dist = nil, rad
        local mypos = plr.Character and plr.Character.HumanoidRootPart.Position
        if not mypos then return nil end
        for _, p in ipairs(game.Players:GetPlayers()) do
            if p ~= plr and p.Character then
                local hrp = p.Character:FindFirstChild("HumanoidRootPart")
                local hum = p.Character:FindFirstChildOfClass("Humanoid")
                if hrp and hum and hum.Health > 0 then
                    if state.silent_visible then
                        local _, onscreen = cam:WorldToViewportPoint(hrp.Position)
                        if not onscreen and not state.silent_wallbang then
                            continue
                        end
                    end
                    local d = (hrp.Position - mypos).Magnitude
                    if d < dist then dist = d; closest = p end
                end
            end
        end
        return closest
    end

    local function getclosestfov(rad, fov)
        local closest, best = nil, math.huge
        local mypos = plr.Character and plr.Character.HumanoidRootPart.Position
        if not mypos then return nil end
        local sc = vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)
        for _, p in ipairs(game.Players:GetPlayers()) do
            if p ~= plr and p.Character then
                local hrp = p.Character:FindFirstChild("HumanoidRootPart")
                local hum = p.Character:FindFirstChildOfClass("Humanoid")
                if hrp and hum and hum.Health > 0 then
                    local d3 = (hrp.Position - mypos).Magnitude
                    if d3 <= rad then
                        local sp, onscreen = cam:WorldToViewportPoint(hrp.Position)
                        if onscreen then
                            local d2 = (vector2.new(sp.X, sp.Y) - sc).Magnitude
                            if d2 <= fov and d2 < best then
                                best = d2
                                closest = p
                            end
                        end
                    end
                end
            end
        end
        return closest
    end

    local function predict(part, vel)
        if not part then return part end
        local mypos = cam.CFrame.Position
        local dist = (part.Position - mypos).Magnitude
        local t = dist / 200
        local k = state.prediction_auto and (0.16 + math.clamp(dist / 5000, 0, 0.1)) or state.prediction
        return part.Position + (vel or vector3.zero) * t * k
    end

    local function silentaim()
        if not state.silent then return end
        local target = state.silent_fov > 0 and getclosestfov(500, state.silent_fov) or getclosest(500)
        state.silent_target = target
        if target and target.Character and target.Character:FindFirstChild("Head") then
            local head = target.Character.Head
            local vel = head.AssemblyLinearVelocity or vector3.zero
            local pos = predict(head, vel)
            local goal = cframe.lookAt(cam.CFrame.Position, pos)
            if state.silent_smooth <= 1 then
                cam.CFrame = goal
            else
                cam.CFrame = cam.CFrame:Lerp(goal, 1 / state.silent_smooth)
            end
        end
    end

    local function killaura()
        if not state.killaura then return end
        if tick() - (state.aura_last or 0) < state.aura_delay then return end
        state.aura_last = tick()
        local targets = {}
        local mypos = plr.Character and plr.Character.HumanoidRootPart.Position
        if not mypos then return end
        for _, p in ipairs(game.Players:GetPlayers()) do
            if p ~= plr and p.Character then
                local hrp = p.Character:FindFirstChild("HumanoidRootPart")
                local hum = p.Character:FindFirstChildOfClass("Humanoid")
                if hrp and hum and hum.Health > 0 then
                    if (hrp.Position - mypos).Magnitude <= state.aura_radius then
                        table.insert(targets, p)
                    end
                end
            end
        end
        if #targets == 0 then return end
        if state.aura_rotate and targets[1].Character then
            local head = targets[1].Character:FindFirstChild("Head")
            if head then
                cam.CFrame = cframe.lookAt(cam.CFrame.Position, head.Position)
            end
        end
        local tool = plr.Character:FindFirstChildOfClass("Tool")
        if tool then
            pcall(function()
                tool:Activate()
                if tool:FindFirstChild("Swing") then tool.Swing:FireServer() end
            end)
            if state.aura_multihit then
                for _, t in ipairs(targets) do
                    pcall(function()
                        tool:Activate()
                        if tool:FindFirstChild("Swing") then tool.Swing:FireServer() end
                    end)
                end
            end
        end
    end

    local function getweapons()
        local folder = workspace:FindFirstChild("Weapons") or workspace:FindFirstChild("Ignored")
        if not folder then return {} end
        local list = {}
        for _, g in ipairs(folder:GetChildren()) do
            if g:FindFirstChild("Handle") then
                if state.autofarm_knife_only then
                    if g.Name:lower():find("knife") then table.insert(list, g) end
                else
                    table.insert(list, g)
                end
            end
        end
        return list
    end

    local function autofarm()
        if not state.autofarm then return end
        local hrp = plr.Character and plr.Character.HumanoidRootPart
        if not hrp then return end
        local weapons = getweapons()
        if #weapons == 0 then return end
        local closest, dist = nil, math.huge
        for _, g in ipairs(weapons) do
            local d = (g.Handle.Position - hrp.Position).Magnitude
            if d < dist then dist = d; closest = g end
        end
        if closest then
            hrp.CFrame = hrp.CFrame + (closest.Handle.Position - hrp.Position).Unit * state.autofarm_speed
        end
    end

    local function esp()
        if not state.esp then
            for p, data in pairs(esp_cache) do
                pcall(function()
                    if data.highlight then data.highlight:Destroy() end
                    if data.billboard then data.billboard:Destroy() end
                end)
                esp_cache[p] = nil
            end
            return
        end
        for _, p in ipairs(game.Players:GetPlayers()) do
            if p ~= plr and p.Character then
                local existing = esp_cache[p]
                if not existing or not existing.highlight or not existing.highlight.Parent then
                    local theme = gettheme()
                    local hl = instance.new("Highlight")
                    hl.Name = "shirato_esp"
                    hl.FillColor = theme.main
                    hl.OutlineColor = color3.fromrgb(255, 255, 255)
                    hl.FillTransparency = 0.55
                    hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                    hl.Parent = p.Character

                    local head = p.Character:FindFirstChild("Head")
                    local bg = nil
                    if head then
                        bg = instance.new("BillboardGui")
                        bg.Name = "shirato_info"
                        bg.Size = udim2.new(0, 220, 0, 60)
                        bg.StudsOffset = vector3.new(0, 3, 0)
                        bg.AlwaysOnTop = true
                        bg.Parent = head

                        local nm = instance.new("TextLabel")
                        nm.Name = "namelabel"
                        nm.Size = udim2.new(1, 0, 0, 20)
                        nm.BackgroundTransparency = 1
                        nm.TextColor3 = color3.fromrgb(255, 255, 255)
                        nm.TextStrokeTransparency = 0
                        nm.Font = Enum.Font.GothamBold
                        nm.TextSize = 14
                        nm.Text = p.Name
                        nm.Parent = bg

                        local hlbl = instance.new("TextLabel")
                        hlbl.Name = "healthlabel"
                        hlbl.Size = udim2.new(1, 0, 0, 18)
                        hlbl.Position = udim2.new(0, 0, 0, 20)
                        hlbl.BackgroundTransparency = 1
                        hlbl.TextColor3 = color3.fromrgb(80, 220, 120)
                        hlbl.TextStrokeTransparency = 0
                        hlbl.Font = Enum.Font.Gotham
                        hlbl.TextSize = 12
                        hlbl.Parent = bg

                        local dlbl = instance.new("TextLabel")
                        dlbl.Name = "distlabel"
                        dlbl.Size = udim2.new(1, 0, 0, 18)
                        dlbl.Position = udim2.new(0, 0, 0, 38)
                        dlbl.BackgroundTransparency = 1
                        dlbl.TextColor3 = color3.fromrgb(200, 200, 200)
                        dlbl.TextStrokeTransparency = 0
                        dlbl.Font = Enum.Font.Gotham
                        dlbl.TextSize = 12
                        dlbl.Parent = bg
                    end
                    esp_cache[p] = {highlight = hl, billboard = bg}
                end
                local data = esp_cache[p]
                if data and data.billboard then
                    local hum = p.Character:FindFirstChildOfClass("Humanoid")
                    local hrp = p.Character:FindFirstChild("HumanoidRootPart")
                    if hum and hrp and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
                        local d = (hrp.Position - plr.Character.HumanoidRootPart.Position).Magnitude
                        local nm = data.billboard:FindFirstChild("namelabel")
                        local hlbl = data.billboard:FindFirstChild("healthlabel")
                        local dlbl = data.billboard:FindFirstChild("distlabel")
                        if nm then nm.Text = state.esp_name and p.Name or "" end
                        if hlbl then hlbl.Text = state.esp_health and string.format("hp: %d", math.floor(hum.Health)) or "" end
                        if dlbl then dlbl.Text = state.esp_distance and string.format("%d studs", math.floor(d)) or "" end
                    end
                end
            end
        end
        for p, data in pairs(esp_cache) do
            if not p.Character or not p.Character.Parent then
                pcall(function()
                    if data.highlight then data.highlight:Destroy() end
                    if data.billboard then data.billboard:Destroy() end
                end)
                esp_cache[p] = nil
            end
        end
    end

    local function gunesp()
        if not state.gunesp then
            for g, hl in pairs(gun_cache) do
                pcall(function() hl:Destroy() end)
            end
            gun_cache = {}
            return
        end
        local folder = workspace:FindFirstChild("Weapons")
        if not folder then return end
        for _, g in ipairs(folder:GetChildren()) do
            if g:FindFirstChild("Handle") and not gun_cache[g] then
                local isknife = g.Name:lower():find("knife")
                if (isknife and state.gunesp_knife) or (not isknife and state.gunesp_gun) then
                    local hl = instance.new("Highlight")
                    hl.FillColor = isknife and color3.fromrgb(255, 200, 60) or color3.fromrgb(0, 200, 255)
                    hl.OutlineColor = color3.fromrgb(255, 255, 255)
                    hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                    hl.Parent = g
                    gun_cache[g] = hl
                end
            end
        end
        for g, hl in pairs(gun_cache) do
            if not g.Parent then
                pcall(function() hl:Destroy() end)
                gun_cache[g] = nil
            end
        end
    end

    local function noclip()
        if not state.noclip then return end
        if plr.Character then
            for _, p in ipairs(plr.Character:GetDescendants()) do

                hrp.AssemblyLinearVelocity = vector3.zero

        end
    end

        end
    end

    local function makeslider(parent, name, key, min, max, def, decimals)
        decimals = decimals or 2
        local holder = instance.new("Frame")
        holder.Size = udim2.new(1, 0, 0, 50)
        holder.BackgroundColor3 = gettheme().bg3
        holder.BorderSizePixel = 0
        holder.Parent = parent

        local hc = instance.new("UICorner")
        hc.CornerRadius = udim.new(0, 6)
        hc.Parent = holder

        local lbl = instance.new("TextLabel")
        lbl.Size = udim2.new(1, -10, 0, 20)
        lbl.Position = udim2.new(0, 5, 0, 2)
        lbl.BackgroundTransparency = 1
        lbl.Text = name .. ": " .. string.format("%." .. decimals .. "f", def)
        lbl.TextColor3 = color3.fromrgb(230, 230, 230)
        lbl.Font = Enum.Font.Gotham
        lbl.TextSize = 13
        lbl.TextXAlignment = Enum.TextXAlignment.Left
        lbl.Parent = holder

        local bar = instance.new("Frame")
        bar.Size = udim2.new(1, -20, 0, 6)
        bar.Position = udim2.new(0, 10, 0, 32)
        bar.BackgroundColor3 = color3.fromrgb(60, 60, 70)
        bar.BorderSizePixel = 0
        bar.Parent = holder

        local bc = instance.new("UICorner")
        bc.CornerRadius = udim.new(1, 0)
        bc.Parent = bar

        local fill = instance.new("Frame")
        fill.Size = udim2.new((def - min) / (max - min), 0, 1, 0)
        fill.BackgroundColor3 = gettheme().main
        fill.BorderSizePixel = 0
        fill.Parent = bar

        local fc = instance.new("UICorner")
        fc.CornerRadius = udim.new(1, 0)
        fc.Parent = fill

        local dragging = false

        local function update(input)
            local rel = math.clamp((input.Position.X - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
            local val = min + (max - min) * rel
            fill.Size = udim2.new(rel, 0, 1, 0)
            lbl.Text = name .. ": " .. string.format("%." .. decimals .. "f", val)
            state[key] = val
        end

        bar.InputBegan:Connect(function(i)
            if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
                dragging = true
                update(i)
            end
        end)
        bar.InputEnded:Connect(function(i)
            if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
                dragging = false
            end
        end)
        uis.InputChanged:Connect(function(i)
            if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
                update(i)
            end
        end)
        uis.InputEnded:Connect(function(i)
            if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
                dragging = false
            end
        end)
    end

        local holder = instance.new("Frame")
        holder.Size = udim2.new(1, 0, 0, 32)

        end)
    end

        table.insert(frametimes, now)
        for i = #frametimes, 1, -1 do

        end

local function checkkey(input)
    if input == VALID_KEY then
        keystatus.Text = "key accepted"
        keystatus.TextColor3 = color3.fromrgb(80, 220, 120)
        task.wait(0.5)
        keygui:Destroy()
        task.spawn(loadmain)
    else
        keystatus.Text = "invalid key"
        keystatus.TextColor3 = color3.fromrgb(255, 80, 80)
        keybox.Text = ""
    end
end

    checkkey(keybox.Text)
end)

    if enter then checkkey(keybox.Text) end
