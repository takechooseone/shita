local state = {
    silent = false,
    prediction = 0.165,
    killaura = false,
    aura_radius = 18,
    aura_delay = 0.1,
    autofarm = false,
    autofarm_speed = 5,
    esp = false,
    esp_name = true,
    esp_health = true,
    gunesp = false,
    fullbright = false,
    noclip = false,
    walkspeed = 16,
    jumppower = 50,
    anti_fling = false,
    anti_void = false,
}

local esp_cache = {}
local gun_cache = {}
local aura_last = 0

local function getclosest(rad)
    local closest, dist = nil, rad or math.huge
    local mypos = plr.Character and plr.Character.HumanoidRootPart and plr.Character.HumanoidRootPart.Position
    if not mypos then return nil end
    for _, p in ipairs(game.Players:GetPlayers()) do
        if p ~= plr and p.Character then
            local hrp = p.Character:FindFirstChild("HumanoidRootPart")
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            if hrp and hum and hum.Health > 0 then
                local d = (hrp.Position - mypos).Magnitude
                if d < dist then dist = d; closest = p end
            end
        end
    end
    return closest
end

local function silentaim()
    if not state.silent then return end
    local target = getclosest(500)
    state.silent_target = target
    if target and target.Character and target.Character:FindFirstChild("Head") then
        local head = target.Character.Head
        local hrp = target.Character:FindFirstChild("HumanoidRootPart")
        if hrp then
            local vel = hrp.AssemblyLinearVelocity or Vector3.new()
            local dist = (head.Position - cam.CFrame.Position).Magnitude
            local t = dist / 200
            local pos = head.Position + vel * t * state.prediction
            cam.CFrame = CFrame.lookAt(cam.CFrame.Position, pos)
        end
    end
end

local function killaura()
    if not state.killaura then return end
    if tick() - aura_last < state.aura_delay then return end
    aura_last = tick()
    local target = getclosest(state.aura_radius)
    if not target then return end
    local tool = plr.Character:FindFirstChildOfClass("Tool")
    if tool then
        pcall(function() tool:Activate() end)
    end
end

local function getweapons()
    local folder = workspace:FindFirstChild("Weapons") or workspace:FindFirstChild("Ignored")
    if not folder then return {} end
    local list = {}
    for _, g in ipairs(folder:GetChildren()) do
        if g:FindFirstChild("Handle") then table.insert(list, g) end
    end
    return list
end

local function autofarm()
    if not state.autofarm then return end
    local hrp = plr.Character and plr.Character.HumanoidRootPart
    if not hrp then return end
    local weapons = getweapons()
    if #weapons == 0 then return end
    local closest, dist = nil, math.huge
    for _, g in ipairs(weapons) do
        local d = (g.Handle.Position - hrp.Position).Magnitude
        if d < dist then dist = d; closest = g end
    end
    if closest then
        local dir = (closest.Handle.Position - hrp.Position)
        hrp.CFrame = hrp.CFrame + dir.Unit * state.autofarm_speed
    end
end

local function esp()
    if not state.esp then
        for p, data in pairs(esp_cache) do
            pcall(function()
                if data.hl then data.hl:Destroy() end
                if data.bg then data.bg:Destroy() end
            end)
            esp_cache[p] = nil
        end
        return
    end
    for _, p in ipairs(game.Players:GetPlayers()) do
        if p ~= plr and p.Character then
            local data = esp_cache[p]
            if not data or not data.hl or not data.hl.Parent then
                local hl = Instance.new("Highlight")
                hl.FillColor = Color3.fromRGB(255, 40, 40)
                hl.OutlineColor = Color3.fromRGB(255, 255, 255)
                hl.FillTransparency = 0.55
                hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                hl.Parent = p.Character

                local head = p.Character:FindFirstChild("Head")
                local bg = nil
                if head then
                    bg = Instance.new("BillboardGui")
                    bg.Size = UDim2.new(0, 220, 0, 60)
                    bg.StudsOffset = Vector3.new(0, 3, 0)
                    bg.AlwaysOnTop = true
                    bg.Parent = head

                    local nm = Instance.new("TextLabel")
                    nm.Name = "nm"
                    nm.Size = UDim2.new(1, 0, 0, 20)
                    nm.BackgroundTransparency = 1
                    nm.TextColor3 = Color3.fromRGB(255, 255, 255)
                    nm.TextStrokeTransparency = 0
                    nm.Font = Enum.Font.GothamBold
                    nm.TextSize = 14
                    nm.Text = p.Name
                    nm.Parent = bg

                    local hpl = Instance.new("TextLabel")
                    hpl.Name = "hp"
                    hpl.Size = UDim2.new(1, 0, 0, 18)
                    hpl.Position = UDim2.new(0, 0, 0, 20)
                    hpl.BackgroundTransparency = 1
                    hpl.TextColor3 = Color3.fromRGB(80, 220, 120)
                    hpl.TextStrokeTransparency = 0
                    hpl.Font = Enum.Font.Gotham
                    hpl.TextSize = 12
                    hpl.Parent = bg

                    local dl = Instance.new("TextLabel")
                    dl.Name = "dl"
                    dl.Size = UDim2.new(1, 0, 0, 18)
                    dl.Position = UDim2.new(0, 0, 0, 38)
                    dl.BackgroundTransparency = 1
                    dl.TextColor3 = Color3.fromRGB(200, 200, 200)
                    dl.TextStrokeTransparency = 0
                    dl.Font = Enum.Font.Gotham
                    dl.TextSize = 12
                    dl.Parent = bg
                end
                esp_cache[p] = {hl = hl, bg = bg}
            end
            local d = esp_cache[p]
            if d and d.bg and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
                local hum = p.Character:FindFirstChildOfClass("Humanoid")
                local hrp = p.Character:FindFirstChild("HumanoidRootPart")
                if hum and hrp then
                    local dist = (hrp.Position - plr.Character.HumanoidRootPart.Position).Magnitude
                    local nm = d.bg:FindFirstChild("nm")
                    local hpl = d.bg:FindFirstChild("hp")
                    local dl = d.bg:FindFirstChild("dl")
                    if nm then nm.Text = state.esp_name and p.Name or "" end
                    if hpl then hpl.Text = state.esp_health and string.format("hp: %d", math.floor(hum.Health)) or "" end
                    if dl then dl.Text = string.format("%d studs", math.floor(dist)) end
                end
            end
        end
    end
end

local function gunesp()
    if not state.gunesp then
        for g, hl in pairs(gun_cache) do
            pcall(function() hl:Destroy() end)
        end
        gun_cache = {}
        return
    end
    local folder = workspace:FindFirstChild("Weapons")
    if not folder then return end
    for _, g in ipairs(folder:GetChildren()) do
        if g:FindFirstChild("Handle") and not gun_cache[g] then
            local hl = Instance.new("Highlight")
            hl.FillColor = Color3.fromRGB(0, 200, 255)
            hl.OutlineColor = Color3.fromRGB(255, 255, 255)
            hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            hl.Parent = g
            gun_cache[g] = hl
        end
    end
    for g, hl in pairs(gun_cache) do
        if not g.Parent then
            pcall(function() hl:Destroy() end)
            gun_cache[g] = nil
        end
    end
end

local function fullbright()
    if state.fullbright then
        game.Lighting.Ambient = Color3.fromRGB(200, 200, 200)
        game.Lighting.Brightness = 2
        game.Lighting.ClockTime = 12
        game.Lighting.FogEnd = 100000
        game.Lighting.GlobalShadows = false
    end
end

local function noclip()
    if not state.noclip then return end
    if plr.Character then
        for _, p in ipairs(plr.Character:GetDescendants()) do
            if p:IsA("BasePart") then p.CanCollide = false end
        end
    end
end

local function antifling()
    if not state.anti_fling then return end
    if plr.Character then
        local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
        if hrp and hrp.AssemblyLinearVelocity.Magnitude > 200 then
            hrp.AssemblyLinearVelocity = Vector3.new()
            hrp.AssemblyAngularVelocity = Vector3.new()
        end
    end
end

local function antivoid()
    if not state.anti_void then return end
    if plr.Character then
        local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
        if hrp and hrp.Position.Y < -50 then
            hrp.CFrame = CFrame.new(0, 50, 0)
        end
    end
end

local function applyspeed()
    if plr.Character and plr.Character:FindFirstChildOfClass("Humanoid") then
        plr.Character.Humanoid.WalkSpeed = state.walkspeed
        plr.Character.Humanoid.JumpPower = state.jumppower
        plr.Character.Humanoid.UseJumpPower = true
    end
end

plr.CharacterAdded:Connect(function()
    task.wait(1)
    applyspeed()
end)

local tabs = {}

local function selecttab(name)
    for tn, td in pairs(tabs) do
        if tn == name then
            td.button.BackgroundColor3 = Color3.fromRGB(80, 30, 30)
            td.button.TextColor3 = Color3.fromRGB(255, 80, 80)
        else
            td.button.BackgroundColor3 = Color3.fromRGB(30, 30, 38)
            td.button.TextColor3 = Color3.fromRGB(200, 200, 200)
        end
    end
    for _, child in ipairs(content:GetChildren()) do
        if child:IsA("GuiObject") and child.Name:sub(1, 6) == "tab_c_" then
            child.Visible = child.Name == "tab_c_" .. name
        end
    end
end

local function createtab(name)
    local button = Instance.new("TextButton")
    button.Size = UDim2.new(0, 90, 1, 0)
    button.BackgroundColor3 = Color3.fromRGB(30, 30, 38)
    button.BorderSizePixel = 0
    button.Text = name
    button.TextColor3 = Color3.fromRGB(200, 200, 200)
    button.Font = Enum.Font.GothamBold
    button.TextSize = 13
    button.Parent = tabbar

    local bc = Instance.new("UICorner")
    bc.CornerRadius = UDim.new(0, 6)
    bc.Parent = button

    local container = Instance.new("Frame")
    container.Name = "tab_c_" .. name
    container.Size = UDim2.new(1, 0, 0, 0)
    container.AutomaticSize = Enum.AutomaticSize.Y
    container.BackgroundTransparency = 1
    container.BorderSizePixel = 0
    container.Visible = false
    container.Parent = content

    local cl = Instance.new("UIListLayout")
    cl.Padding = UDim.new(0, 6)
    cl.SortOrder = Enum.SortOrder.LayoutOrder
    cl.Parent = container

    button.MouseButton1Click:Connect(function()
        selecttab(name)
    end)

    tabs[name] = {button = button, container = container, layout = cl}
    return container
end

end

local function makeslider(parent, name, key, min, max, def)
    local holder = Instance.new("Frame")
    holder.Size = UDim2.new(1, 0, 0, 50)
    holder.BackgroundColor3 = Color3.fromRGB(30, 30, 38)
    holder.BorderSizePixel = 0
    holder.Parent = parent

    local hc = Instance.new("UICorner")
    hc.CornerRadius = UDim.new(0, 6)
    hc.Parent = holder

    local lbl = Instance.new("TextLabel")
    lbl.Size = UDim2.new(1, -10, 0, 20)
    lbl.Position = UDim2.new(0, 5, 0, 2)
    lbl.BackgroundTransparency = 1
    lbl.Text = name .. ": " .. def
    lbl.TextColor3 = Color3.fromRGB(230, 230, 230)
    lbl.Font = Enum.Font.Gotham
    lbl.TextSize = 13
    lbl.TextXAlignment = Enum.TextXAlignment.Left
    lbl.Parent = holder

    local bar = Instance.new("Frame")
    bar.Size = UDim2.new(1, -20, 0, 6)
    bar.Position = UDim2.new(0, 10, 0, 32)
    bar.BackgroundColor3 = Color3.fromRGB(60, 60, 70)
    bar.BorderSizePixel = 0
    bar.Parent = holder

    local bc = Instance.new("UICorner")
    bc.CornerRadius = UDim.new(1, 0)
    bc.Parent = bar

    local fill = Instance.new("Frame")
    fill.Size = UDim2.new((def - min) / (max - min), 0, 1, 0)
    fill.BackgroundColor3 = Color3.fromRGB(255, 40, 40)
    fill.BorderSizePixel = 0
    fill.Parent = bar

    local fc = Instance.new("UICorner")
    fc.CornerRadius = UDim.new(1, 0)
    fc.Parent = fill

    local dragging = false

    local function update(input)
        local rel = math.clamp((input.Position.X - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
        local val = min + (max - min) * rel
        fill.Size = UDim2.new(rel, 0, 1, 0)
        lbl.Text = name .. ": " .. string.format("%.2f", val)
        state[key] = val
    end

    bar.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            update(i)
        end
    end)
    bar.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    uis.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            update(i)
        end
    end)
    uis.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end
